import subprocess, os
FF = r"C:\Users\1234d\AppData\Local\Microsoft\WinGet\Packages\Gyan.FFmpeg_Microsoft.Winget.Source_8wekyb3d8bbwe\ffmpeg-9.0.2-full_build\bin\ffmpeg.EXE"
D = r"C:\Users\1234d\Documents\AmbientChannel\08_rainy_cafe"
os.chdir(D); os.makedirs("units2", exist_ok=True)
log = open("units2_log.txt", "w")
ENC = ["-c:v","libx265","-b:v","7M","-maxrate","9M","-bufsize","14M","-preset","medium","-tag:v","hvc1","-pix_fmt","yuv420p","-r","24",
       "-x265-params","log-level=error:keyint=48:min-keyint=48:scenecut=0"]
UP = "scale=2560:1440:flags=lanczos,unsharp=5:5:0.6,format=yuv420p"
# name: (source, start frame, end frame) ; 4-frame blend at the join
src = {"v1_wide":("L1_wide.mp4",1,185),"v2_window":("L2_window.mp4",0,184),"v3_counter":("L3_counter.mp4",0,184),
       "v4_corner":("L4_corner.mp4",2,183),"v5_night":("L5_night.mp4",0,184)}
X = 4
def run(cmd):
    r = subprocess.run(cmd, capture_output=True, text=True, creationflags=0x08000000)
    log.write(cmd[-1] + " rc=%d %s\n" % (r.returncode, r.stderr[-300:])); log.flush()
for n,(f,S,E) in src.items():
    fc = ("[0:v]fps=24,split=3[a][b][c];[a]trim=start_frame=%d:end_frame=%d,setpts=PTS-STARTPTS[m];"
          "[b]trim=start_frame=%d:end_frame=%d,setpts=PTS-STARTPTS[t];[c]trim=start_frame=%d:end_frame=%d,setpts=PTS-STARTPTS[h];"
          "[t][h]xfade=transition=fade:duration=%.4f:offset=0[x];[m][x]concat=n=2:v=1:a=0,%s[v]") % (S+X,E-X,E-X,E,S,S+X,X/24.0,UP)
    run([FF,"-y","-v","error","-i",f,"-filter_complex",fc,"-map","[v]"]+ENC+["units2/%s.mp4"%n])
pairs = [("v1_wide","v2_window"),("v2_window","v3_counter"),("v3_counter","v4_corner"),("v4_corner","v1_wide"),
         ("v4_corner","v5_night"),("v5_night","v2_window"),("v2_window","v5_night"),("v5_night","v4_corner"),("v1_wide","v5_night")]
FP = FF.replace("ffmpeg.EXE","ffprobe.exe")
def dur(p): return float(subprocess.run([FP,"-v","error","-show_entries","format=duration","-of","csv=p=0",p],capture_output=True,text=True,creationflags=0x08000000).stdout)
for a,b in pairs:
    fc = "[0:v][1:v]xfade=transition=fade:duration=2.5:offset=%.3f,format=yuv420p[v]" % (dur("units2/%s.mp4"%a)-2.5)
    run([FF,"-y","-v","error","-i","units2/%s.mp4"%a,"-i","units2/%s.mp4"%b,"-filter_complex",fc,"-map","[v]"]+ENC+["units2/T_%s__%s.mp4"%(a,b)])
log.write("DONE\n"); log.close()
